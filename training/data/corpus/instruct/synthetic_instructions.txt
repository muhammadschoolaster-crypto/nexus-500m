### Instruction:
Write a Python function that reverses a string without slicing.
### Input:
reverse('nexus')
### Response:
def reverse(s: str) -> str:
    result = []
    for ch in s:
        result.insert(0, ch)
    return ''.join(result)

print(reverse('nexus'))  # 'suxen'
